cmake_minimum_required(VERSION 3.25)
project(BTKaraoke VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(FetchContent)
find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Widgets)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Widgets Test)
set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
set(SMTG_RUN_VST_VALIDATOR OFF CACHE BOOL "" FORCE)
set(SMTG_CREATE_MODULE_INFO OFF CACHE BOOL "" FORCE)
FetchContent_Declare(vst3 GIT_REPOSITORY https://github.com/steinbergmedia/vst3sdk.git GIT_TAG 9fad9770f2ae8542ab1a548a68c1ad1ac690abe0 GIT_SUBMODULES base cmake pluginterfaces public.sdk GIT_SHALLOW FALSE)
FetchContent_MakeAvailable(vst3)
FetchContent_Declare(miniaudio GIT_REPOSITORY https://github.com/mackron/miniaudio.git GIT_TAG 350784a9467a79d0fa65802132668e5afbcf3777 SOURCE_SUBDIR __header_only)
FetchContent_Declare(tsf GIT_REPOSITORY https://github.com/schellingb/TinySoundFont.git GIT_TAG 853a0a171759f1ddba0de1442133a75912bbeffa SOURCE_SUBDIR __header_only)
FetchContent_MakeAvailable(miniaudio tsf)
add_library(btcore src/core/Midi.cpp src/audio/MidiOutput.cpp src/audio/Effect.cpp src/audio/Engine.cpp src/plugins/VstEffect.cpp)
target_include_directories(btcore PUBLIC src PRIVATE ${miniaudio_SOURCE_DIR} ${tsf_SOURCE_DIR})
target_link_libraries(btcore PUBLIC Qt${QT_VERSION_MAJOR}::Widgets sdk_hosting)
if(WIN32)
 target_sources(btcore PRIVATE ${vst3_SOURCE_DIR}/public.sdk/source/vst/hosting/module_win32.cpp)
 target_link_libraries(btcore PRIVATE ole32 uuid winmm shell32)
 target_compile_options(btcore PUBLIC /utf-8)
 target_compile_definitions(btcore PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
else()
 target_sources(btcore PRIVATE ${vst3_SOURCE_DIR}/public.sdk/source/vst/hosting/module_linux.cpp)
 target_link_libraries(btcore PRIVATE dl pthread)
endif()
add_executable(BTKaraoke WIN32 src/main.cpp src/ui/MainWindow.cpp)
target_link_libraries(BTKaraoke PRIVATE btcore)
set_target_properties(BTKaraoke PROPERTIES AUTOMOC ON)
if(WIN32)
 set_target_properties(BTKaraoke PROPERTIES OUTPUT_NAME BTKaraokeApp)
 add_executable(BTLauncher WIN32 src/launcher_win.cpp resources/windows.rc)
 set_target_properties(BTLauncher PROPERTIES OUTPUT_NAME BTKaraoke MSVC_RUNTIME_LIBRARY MultiThreaded)
 target_include_directories(BTLauncher PRIVATE resources)
 target_compile_options(BTLauncher PRIVATE /utf-8)
 target_compile_definitions(BTLauncher PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
 target_link_libraries(BTLauncher PRIVATE user32)
 add_dependencies(BTKaraoke BTLauncher)
 install(TARGETS BTLauncher RUNTIME DESTINATION .)
endif()

add_executable(bt_tests tests/CoreTests.cpp)
target_link_libraries(bt_tests PRIVATE btcore Qt${QT_VERSION_MAJOR}::Test)
set_target_properties(bt_tests PROPERTIES AUTOMOC ON)
enable_testing()
add_test(NAME bt_core COMMAND bt_tests -o core-results.txt,txt -o -,txt)
set_tests_properties(bt_core PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
install(TARGETS BTKaraoke RUNTIME DESTINATION .)
target_sources(btcore PRIVATE ${vst3_SOURCE_DIR}/public.sdk/source/vst/hosting/plugprovider.cpp ${vst3_SOURCE_DIR}/public.sdk/source/common/memorystream.cpp)
smtg_add_vst3plugin(bt_test_vst3 tests/TestPlugin.cpp)
target_link_libraries(bt_test_vst3 PRIVATE sdk)
if(WIN32)
 set(BT_TEST_PACKAGE "$<TARGET_FILE:bt_test_vst3>")
else()
 get_target_property(BT_TEST_PACKAGE bt_test_vst3 SMTG_PLUGIN_PACKAGE_PATH)
endif()
add_dependencies(bt_tests bt_test_vst3)
set_tests_properties(bt_core PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;BT_TEST_VST3=${BT_TEST_PACKAGE}")

if(WIN32)
 target_sources(bt_test_vst3 PRIVATE ${vst3_SOURCE_DIR}/public.sdk/source/main/dllmain.cpp)
else()
 target_sources(bt_test_vst3 PRIVATE ${vst3_SOURCE_DIR}/public.sdk/source/main/linuxmain.cpp)
endif()
set_target_properties(BTKaraoke PROPERTIES AUTORCC ON)
target_sources(BTKaraoke PRIVATE resources/app.qrc)
if(WIN32)
 enable_language(RC)
 target_sources(BTKaraoke PRIVATE resources/windows.rc)
 target_include_directories(BTKaraoke PRIVATE resources)
endif()
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/licenses")
configure_file("${vst3_SOURCE_DIR}/LICENSE.txt" "${CMAKE_BINARY_DIR}/licenses/VST3-SDK.txt" COPYONLY)
configure_file("${miniaudio_SOURCE_DIR}/LICENSE" "${CMAKE_BINARY_DIR}/licenses/miniaudio.txt" COPYONLY)
configure_file("${tsf_SOURCE_DIR}/LICENSE" "${CMAKE_BINARY_DIR}/licenses/TinySoundFont.txt" COPYONLY)

add_test(NAME bt_safe_startup COMMAND BTKaraoke --smoke-test --safe-mode)
add_test(NAME bt_normal_startup COMMAND BTKaraoke --startup-test)
set_tests_properties(bt_safe_startup bt_normal_startup PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
